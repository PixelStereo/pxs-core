function(pxs_core_test name)
    add_executable(${name}_test ${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE ${ARGN})
    target_include_directories(${name}_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(${name}_test PRIVATE PXS_GOLDEN_FILE="${CMAKE_CURRENT_SOURCE_DIR}/golden/vectors.tsv")
    pxs_core_harden(${name}_test)
    if(MSVC)
        target_compile_definitions(${name}_test PRIVATE _CRT_SECURE_NO_WARNINGS) # getenv in the robustness test
    endif()
    add_test(NAME ${name} COMMAND ${name}_test)
endfunction()

# The header is C: compiled as C99, with the warnings on
add_executable(c_api_header_test c_api_header_test.c)
target_link_libraries(c_api_header_test PRIVATE pxs-core::c)
set_target_properties(c_api_header_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE CXX)
if(MSVC)
    target_compile_options(c_api_header_test PRIVATE /W4 $<$<BOOL:${PXS_CORE_WARNINGS_AS_ERRORS}>:/WX>)
else()
    target_compile_options(c_api_header_test PRIVATE -Wall -Wextra -Wpedantic -Wconversion $<$<BOOL:${PXS_CORE_WARNINGS_AS_ERRORS}>:-Werror>)
endif()
if(PXS_CORE_SANITIZE AND NOT MSVC)
    target_compile_options(c_api_header_test PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined)
    target_link_options(c_api_header_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME c_api_header COMMAND c_api_header_test)
